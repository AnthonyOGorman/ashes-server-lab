cmake_minimum_required(VERSION 3.24)
if(WIN32 AND NOT CMAKE_GENERATOR MATCHES "^Visual Studio")
  message(FATAL_ERROR "Use tools/Build-CPP.ps1 or the windows-msvc preset. MSVC is the supported compiler.")
endif()
project(AshesCppLab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(NGHTTP2_INCLUDE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/vendor)
set(NGHTTP2_LIBRARY ${CMAKE_CURRENT_SOURCE_DIR}/vendor/lib/nghttp2.lib)
set(SQLITE_LIBRARY ${CMAKE_CURRENT_SOURCE_DIR}/vendor/lib/sqlite3.lib)
set(CRYPTO_LIBRARY ${CMAKE_CURRENT_SOURCE_DIR}/vendor/lib/crypto.lib)
add_library(ashes_core src/common.cpp src/protocol.cpp src/collision.cpp src/movement.cpp src/contracts.cpp src/inspection.cpp src/inspection_hash.cpp src/live_proofs.cpp src/actor_refresh.cpp)
target_include_directories(ashes_core PUBLIC include vendor ${NGHTTP2_INCLUDE_DIR})
target_compile_definitions(ashes_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(ashes_core PUBLIC ws2_32 bcrypt user32 psapi ${CRYPTO_LIBRARY})
if(MSVC)
 target_compile_options(ashes_core PRIVATE /W4 /EHsc /utf-8)
else()
 target_compile_options(ashes_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(ashes_lab src/main.cpp src/backend.cpp src/lobby.cpp src/stages.cpp src/client.cpp src/initialization.cpp src/speed_sync.cpp src/movement_profile.cpp src/resource_profile.cpp src/settlement_collision.cpp)
target_link_libraries(ashes_lab PRIVATE ashes_core ${NGHTTP2_LIBRARY} ${SQLITE_LIBRARY} iphlpapi)
add_executable(ashes_inspect src/inspect_main.cpp)
target_link_libraries(ashes_inspect PRIVATE ashes_core)
add_executable(ashes_floor_probe tools/floor_probe.cpp)
target_link_libraries(ashes_floor_probe PRIVATE ashes_core)
add_library(ashes_eos_local SHARED native/eos_connect_local/connect_local.cpp)
set_target_properties(ashes_eos_local PROPERTIES PREFIX "" OUTPUT_NAME LocalEOSConnect)
add_library(ashes_input_adapter SHARED native/input_adapter/adapter.cpp)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/native/input_adapter/adapter.cpp" ADAPTER_SOURCE_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/native/input_adapter/protocol.h" ADAPTER_PROTOCOL_HASH)
string(SHA256 ADAPTER_BUILD_HASH "${ADAPTER_SOURCE_HASH}${ADAPTER_PROTOCOL_HASH}")
string(SUBSTRING "${ADAPTER_BUILD_HASH}" 0 8 ADAPTER_BUILD_ID)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated/build")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/build/build_tag.h" "constexpr uint32_t BUILD_ID=0x${ADAPTER_BUILD_ID};\n")
target_include_directories(ashes_input_adapter PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(ashes_input_adapter PRIVATE user32 advapi32)
set_target_properties(ashes_input_adapter PROPERTIES PREFIX "" OUTPUT_NAME ashes_input_adapter)
add_executable(ashes_input_fixture native/input_adapter/fixture.cpp)
target_link_libraries(ashes_input_fixture PRIVATE user32)
enable_testing()
add_executable(ashes_tests tests/test_core.cpp)
target_link_libraries(ashes_tests PRIVATE ashes_core)
add_test(NAME core_parity COMMAND ashes_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/parity.json)
add_executable(ashes_movement_tests tests/test_movement.cpp)
target_link_libraries(ashes_movement_tests PRIVATE ashes_core)
add_test(NAME movement_compatibility COMMAND ashes_movement_tests ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(ashes_terrain_tests tests/test_terrain.cpp)
target_link_libraries(ashes_terrain_tests PRIVATE ashes_core)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/data/terrain-offline/decoded/manifest.json")
  add_test(NAME offline_terrain COMMAND ashes_terrain_tests ${CMAKE_CURRENT_SOURCE_DIR})
endif()
add_executable(ashes_inspection_hash_tests tests/test_inspection_hash.cpp)
target_link_libraries(ashes_inspection_hash_tests PRIVATE ashes_core)
add_test(NAME executable_identity_cache COMMAND ashes_inspection_hash_tests ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(ashes_client_readiness_tests tests/test_client_readiness.cpp src/client.cpp)
target_link_libraries(ashes_client_readiness_tests PRIVATE ashes_core)
add_test(NAME client_loading_readiness COMMAND ashes_client_readiness_tests ${CMAKE_CURRENT_SOURCE_DIR})
set(ASHES_DEPENDENCY_BIN "C:/msys64/mingw64/bin" CACHE PATH "MSYS2 MINGW64 dependency DLL directory")
foreach(target ashes_lab ashes_tests ashes_movement_tests ashes_terrain_tests ashes_inspection_hash_tests ashes_client_readiness_tests ashes_inspect ashes_floor_probe ashes_input_fixture ashes_input_adapter ashes_eos_local)
  foreach(dll libgcc_s_seh-1.dll libstdc++-6.dll libwinpthread-1.dll libnghttp2-14.dll libsqlite3-0.dll libcrypto-3-x64.dll)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ASHES_DEPENDENCY_BIN}/${dll}" $<TARGET_FILE_DIR:${target}>)
  endforeach()
endforeach()
